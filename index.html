<script>
const state = {
name: "Alex Smith",
age: 0,
bank: 0,
health: 100,
happiness: 100,
smarts: 50,
job: null,
isDead: false
};

const avatars = { baby: "👶", child: "🧒", teen: "🧑", adult: "👨", senior: "👨‍🦳" };

function updateUI() {
document.getElementById('charSub').innerText = `Age ${state.age} • ${getStage()}`;
document.getElementById('charBank').innerText = `$${state.bank.toLocaleString()}`;

// Stats
['health', 'happy', 'smarts'].forEach(s => {
const key = s === 'happy' ? 'happiness' : s;
document.getElementById(`${s}Val`).innerText = `${state[key]}%`;
document.getElementById(`${s}Bar`).style.width = `${state[key]}%`;
});

// Wealth Bar Scaling
const wealthPct = Math.min(100, Math.floor((state.bank / 100000) * 100));
document.getElementById('wealthVal').innerText = `${wealthPct}%`;
document.getElementById('wealthBar').style.width = `${wealthPct}%`;

// Avatar
let emoji = avatars.baby;
if (state.age >= 60) emoji = avatars.senior;
else if (state.age >= 18) emoji = avatars.adult;
else if (state.age >= 12) emoji = avatars.teen;
else if (state.age >= 5) emoji = avatars.child;
document.getElementById('avatar').innerText = emoji;
}

function getStage() {
if (state.age >= 60) return "Senior";
if (state.age >= 18) return "Adult";
if (state.age >= 12) return "Teen";
if (state.age >= 5) return "Child";
return "Baby";
}

function addLog(text, isAge = false) {
const feed = document.getElementById('lifeFeed');
const entry = document.createElement('div');
entry.className = `log-entry ${isAge ? 'age-up' : ''}`;
entry.innerText = text;
feed.appendChild(entry);
feed.scrollTop = feed.scrollHeight;
}

function ageUp() {
if (state.isDead) return;
state.age++;
addLog(`Age ${state.age}: Another year passes.`, true);

// Job Income
if (state.job) {
state.bank += state.job.salary;
addLog(`Earned $${state.job.salary.toLocaleString()} as a ${state.job.title}.`);
}

// Life Events
if (state.age === 6) {
triggerChoice("Elementary School", "It's your first day of school!", [
{ label: "Study hard", action: () => { state.smarts += 15; addLog("You studied hard and made good grades!"); } },
{ label: "Make friends", action: () => { state.happiness += 10; addLog("You focused on making friends."); } }
]);
} else if (state.age === 18 && !state.job) {
triggerChoice("Career Path", "You graduated high school! Choose a path:", [
{ label: "Get a job as Fast Food Worker ($18k/yr)", action: () => { state.job = { title: "Fast Food Worker", salary: 18000 }; } },
{ label: "Go to College ($20k tuition, +Smarts)", action: () => { state.smarts += 30; state.bank -= 20000; state.job = { title: "Software Engineer", salary: 85000 }; addLog("Graduated college and landed a tech job!"); } }
]);
}

// Check health
if (state.health <= 0 || state.age > 100) {
state.isDead = true;
addLog(`🪦 You passed away peacefully at age ${state.age}. Final Wealth: $${state.bank.toLocaleString()}`);
}

updateUI();
}

function triggerChoice(title, desc, options) {
document.getElementById('modalTitle').innerText = title;
document.getElementById('modalDesc').innerText = desc;
const container = document.getElementById('modalOpts');
container.innerHTML = "";

options.forEach(opt => {
const btn = document.createElement('button');
btn.className = "opt-btn";
btn.innerText = opt.label;
btn.onclick = () => {
opt.action();
document.getElementById('choiceModal').classList.remove('active');
updateUI();
};
container.appendChild(btn);
});

document.getElementById('choiceModal').classList.add('active');
}

// Initialize
addLog("Born into the world as Alex Smith.");
updateUI();
</script>
